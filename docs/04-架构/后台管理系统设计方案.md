## 1. 需求分析

### 1.1 业务背景

当前项目已完成C端用户功能，包括猫咪档案管理、体重追踪、疫苗记录、AI健康顾问等。但缺乏**管理员后台**进行：

- **内容管理**：知识指南、计划模板的增删改查
- **用户管理**：查看用户列表、管理用户状态
- **数据监控**：查看平台数据统计、用户行为分析
- **系统配置**：系统参数、通知模板等配置管理

### 1.2 目标用户

| 用户角色 | 职责 |
|---------|------|
| 超级管理员 | 完整的系统管理权限 |
| 内容管理员 | 知识指南、模板管理 |
| 运营人员 | 用户管理、数据查看 |

## 4. 数据库设计

### 4.1 新增管理员表

```prisma
// 管理员表
model Admin {
  id          String   @id @default(cuid())
  username    String   @unique
  password    String
  email       String?  @unique
  role        String   @default("editor") // super, admin, editor
  name        String?
  avatar      String?
  isActive    Boolean  @default(true)
  lastLoginAt DateTime?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}

// 操作日志表
model AdminLog {
  id        String   @id @default(cuid())
  adminId   String?
  admin     Admin?   @relation(fields: [adminId], references: [id])
  action    String   // 操作类型：create, update, delete, login, logout
  module    String   // 操作模块：user, cat, guide, template, config
  targetId  String?  // 目标记录ID
  detail    String?  // 操作详情(JSON)
  ip        String?
  userAgent String?
  createdAt DateTime @default(now())

  @@index([adminId])
  @@index([action])
  @@index([module])
  @@index([createdAt])
}
```

## 7. 权限控制

### 7.1 路由守卫

```typescript
// 管理员路由守卫
router.beforeEach((to, from, next) => {
  const adminStore = useAdminStore()

  // 需要登录的页面
  if (to.meta.requiresAuth && !adminStore.isAuthenticated) {
    next('/admin/login')
    return
  }

  // 需要权限的页面
  if (to.meta.permission && !adminStore.hasPermission(to.meta.permission)) {
    next('/admin/dashboard')
    return
  }

  next()
})
```

## 10. 代码安全性

### 10.1 注意事项

1. **管理员密码加密**：使用bcrypt加密存储
2. **JWT安全**：设置合理过期时间、使用强密钥
3. **输入验证**：所有用户输入必须验证
4. **权限校验**：所有操作必须检查权限
5. **日志审计**：记录所有管理员操作
6. **SQL注入防护**：使用Prisma参数化查询
7. **XSS防护**：对输出内容进行转义
8. **文件上传安全**：限制文件类型和大小

### 10.2 最佳实践

- 使用HTTPS传输
- 定期轮换密钥
- 限制登录尝试次数
- 记录登录失败日志
- 管理员操作需要二次确认（删除等危险操作）

## 12. 前端状态管理详细设计

### 12.1 Admin Store 设计

```typescript
// frontend/src/stores/admin.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { Permission } from '@/types/permissions'

export interface AdminInfo {
  id: string
  username: string
  email?: string
  name?: string
  avatar?: string
  role: 'super' | 'admin' | 'editor'
  permissions: Permission[]
}

export interface LoginCredentials {
  username: string
  password: string
}

export const useAdminStore = defineStore('admin', () => {
  // ========== State ==========
  const token = ref<string>(localStorage.getItem('admin_token') || '')
  const userInfo = ref<AdminInfo | null>(null)
  const permissions = ref<Permission[]>([])
  const isLoading = ref(false)
  const error = ref<string | null>(null)

  // ========== Getters ==========
  const isAuthenticated = computed(() => !!token.value && !!userInfo.value)

  const hasPermission = computed(() => (permission: Permission | Permission[]): boolean => {
    if (!permissions.value.length) return false
    if (Array.isArray(permission)) {
      return permission.some(p => permissions.value.includes(p))
    }
    return permissions.value.includes(permission)
  })

  const hasRole = computed(() => (roles: Array<'super' | 'admin' | 'editor'>): boolean => {
    return userInfo.value?.role ? roles.includes(userInfo.value.role) : false
  })

  const isSuperAdmin = computed(() => userInfo.value?.role === 'super')
  const isAdmin = computed(() => userInfo.value?.role === 'admin' || userInfo.value?.role === 'super')
  const isEditor = computed(() => ['super', 'admin', 'editor'].includes(userInfo.value?.role || ''))

  // ========== Actions ==========
  async function login(credentials: LoginCredentials) {
    isLoading.value = true
    error.value = null
    try {
      const response = await adminApi.login(credentials)
      token.value = response.data.token
      userInfo.value = response.data.admin
      permissions.value = response.data.permissions || []

      // 持久化
      localStorage.setItem('admin_token', token.value)

      return response.data
    } catch (e: any) {
      error.value = e.message || '登录失败'
      throw e
    } finally {
      isLoading.value = false
    }
  }

  async function logout() {
    try {
      await adminApi.logout()
    } catch (e) {
      // 忽略logout错误
    } finally {
      token.value = ''
      userInfo.value = null
      permissions.value = []
      localStorage.removeItem('admin_token')
    }
  }

  async function loadCurrentUser() {
    if (!token.value) return null

    isLoading.value = true
    try {
      const response = await adminApi.getMe()
      userInfo.value = response.data.admin
      permissions.value = response.data.permissions || []
      return response.data
    } catch (e) {
      await logout()
      throw e
    } finally {
      isLoading.value = false
    }
  }

  return {
    token,
    userInfo,
    permissions,
    isLoading,
    error,
    isAuthenticated,
    hasPermission,
    hasRole,
    isSuperAdmin,
    isAdmin,
    isEditor,
    login,
    logout,
    loadCurrentUser,
  }
})
```

```typescript
// frontend/src/composables/useLoading.ts
import { ref } from 'vue'

export function useLoading(initialState = false) {
  const isLoading = ref(initialState)

  const start = () => { isLoading.value = true }
  const stop = () => { isLoading.value = false }

  const wrap = async <T>(fn: () => Promise<T>) => {
    start()
    try {
      return await fn()
    } finally {
      stop()
    }
  }

  return { isLoading, start, stop, wrap }
}
```

**统一错误提示**

```typescript
// frontend/src/utils/message.ts
import { ElMessage } from 'element-plus'

interface ApiError {
  success: false
  error: string
  message: string
  details?: any
}

export function handleApiError(error: ApiError | any, fallback = '操作失败') {
  if (error.success === false) {
    ElMessage.error(error.message || fallback)
    return
  }
  ElMessage.error(error.message || fallback)
}

export function handleApiSuccess(message = '操作成功') {
  ElMessage.success(message)
}

export function handleBatchResult(result: { succeeded: number; failed: number }) {
  if (result.failed === 0) {
    ElMessage.success(`成功操作 ${result.succeeded} 项`)
  } else {
    ElMessage.warning(`成功 ${result.succeeded} 项，失败 ${result.failed} 项`)
  }
}
```

export const validate = (schema: AnyZodObject) =>
  async (req: Request, res: Response, next: NextFunction) => {

        body: req.body,

        params: req.params,

        })
      }
      next(error)
    }
  }

// backend/src/middlewares/authorization.ts
export const requirePermission = (permission: Permission) =>
  async (req: Request, res: Response, next: NextFunction) => {
    if (!req.admin) {
      throw new AppError(401, 'UNAUTHORIZED', '未登录')
    }
    if (!req.admin.permissions.includes(permission)) {
      throw new AppError(403, 'FORBIDDEN', '权限不足')
    }
    next()
  }

export const requireRole = (roles: Role[]) =>
  async (req: Request, res: Response, next: NextFunction) => {
    if (!req.admin) {
      throw new AppError(401, 'UNAUTHORIZED', '未登录')
    }
    if (!roles.includes(req.admin.role)) {
      throw new AppError(403, 'FORBIDDEN', '需要更高权限')
    }
    next()
  }

## 15. 性能优化方案

### 15.1 分页策略

```typescript
// backend/src/common/pagination.ts
export interface PaginationOptions {
  page: number
  pageSize: number
  sortBy?: string
  sortOrder?: 'asc' | 'desc'
}

export async function paginate<T>(
  model: any,
  options: PaginationOptions,
  where: any = {}
): Promise<{ items: T[]; pagination: { page: number; pageSize: number; total: number; totalPages: number } }> {
  const { page, pageSize, sortBy = 'createdAt', sortOrder = 'desc' } = options

  const [items, total] = await Promise.all([
    model.findMany({ where, orderBy: { [sortBy]: sortOrder }, skip: (page - 1) * pageSize, take: pageSize }),
    model.count({ where }),
  ])

  return { items, pagination: { page, pageSize, total, totalPages: Math.ceil(total / pageSize) } }
}
```

export class CacheService {
  private static DEFAULT_TTL = 3600

  static async get<T>(key: string): Promise<T | null> {
    const data = await redis.get(key)
    return data ? JSON.parse(data) : null
  }

  static async set(key: string, value: any, ttl: number = this.DEFAULT_TTL) {
    await redis.setex(key, ttl, JSON.stringify(value))
  }

  static async del(key: string) {
    await redis.del(key)
  }

  static async delPattern(pattern: string) {
    const keys = await redis.keys(pattern)

  }

### 17.1 日志级别规范

```typescript
// backend/src/utils/logger.ts
export enum LogLevel {
  DEBUG = 0,
  INFO = 1,
  WARN = 2,
  ERROR = 3,
}

class Logger {
  private level: LogLevel

  setLevel(level: LogLevel) {
    this.level = level
  }

  debug(message: string, metadata?: any) {
    if (this.level <= LogLevel.DEBUG) {
      console.debug(this.formatEntry({ level: 'DEBUG', message, ...metadata }))
    }
  }

  info(message: string, metadata?: any) {
    if (this.level <= LogLevel.INFO) {
      console.info(this.formatEntry({ level: 'INFO', message, ...metadata }))
    }
  }

  warn(message: string, metadata?: any) {
    if (this.level <= LogLevel.WARN) {
      console.warn(this.formatEntry({ level: 'WARN', message, ...metadata }))
    }
  }

  error(message: string, error?: Error, metadata?: any) {
    if (this.level <= LogLevel.ERROR) {
      console.error(this.formatEntry({ level: 'ERROR', message, error: { message: error?.message, stack: error?.stack?.split('\n')[0] }, ...metadata }))
    }
  }

  private formatEntry(entry: any): string {
    return JSON.stringify({ timestamp: new Date().toISOString(), ...entry })
  }
}

export const logger = new Logger()
logger.setLevel(process.env.NODE_ENV === 'production' ? LogLevel.INFO : LogLevel.DEBUG)
```

### 17.2 敏感信息脱敏策略

```typescript
// backend/src/utils/sanitize.ts
const SENSITIVE_FIELDS = {
  password: '[REDACTED]',
  token: '[REDACTED]',
  authorization: '[REDACTED]',
  email: (val: string) => val ? `${val.slice(0, 2)}***@${val.split('@')[1]}` : '',
  phone: (val: string) => val ? `***${val.slice(-4)}` : '',
  ip: (val: string) => val ? `${val.split('.').slice(0, 2).join('.')}.***` : '',
}

export function sanitizeObject(obj: any, depth = 0): any {
  if (depth > 10 || !obj || typeof obj !== 'object') return obj

  if (Array.isArray(obj)) return obj.map(item => sanitizeObject(item, depth + 1))

  const sanitized: any = {}
  for (const [key, value] of Object.entries(obj)) {
    const lowerKey = key.toLowerCase()
    const isSensitive = Object.keys(SENSITIVE_FIELDS).some(k => lowerKey.includes(k.toLowerCase()))

    if (isSensitive) {
      const sanitizer = SENSITIVE_FIELDS[key]
      sanitized[key] = typeof sanitizer === 'function' ? sanitizer(value) : sanitizer
    } else if (typeof value === 'object') {
      sanitized[key] = sanitizeObject(value, depth + 1)
    } else {
      sanitized[key] = value
    }
  }
  return sanitized
}
```
