# 哈吉咪养成计划 - 开发文档 V3.2

## 项目简介

哈吉咪养成计划是一个专注于猫咪养护的全栈Web应用，提供从幼猫到成年的全程陪伴式养护指导。项目采用前后端分离架构，使用现代化的技术栈构建。

**V3.2 更新**：新增多猫对比功能（支持 2-5 只猫咪的体重趋势、健康状态、基本信息对比）。
**V3.1 更新**：新增体重健康标准系统（品种标准数据、健康状态分析）。
**V3.0 更新**：新增体重趋势可视化图表（ECharts）、增强的认证初始化系统、完善的调试日志体系。

```vue
      >
        <input
          type="checkbox"
          :checked="selected.includes(cat.id)"
          :disabled="!selected.includes(cat.id) && selected.length >= max"
          @change="toggleCat(cat.id)"
        />
        <div class="cat-avatar">
          <img v-if="cat.avatar" :src="getAvatarUrl(cat)" :alt="cat.name" />
          <span v-else class="avatar-placeholder">{{ cat.name?.charAt(0) || '?' }}</span>
        </div>
        <div class="cat-info">
          <div class="cat-name">{{ cat.name }}</div>
          <div class="cat-meta">{{ cat.breed || '未知品种' }} · {{ cat.ageFormatted }}</div>
        </div>
      </label>
    </div>

    <!-- 操作按钮 -->
    <div v-if="cats.length >= 2" class="selector-actions">
      <button @click="selectAll" class="btn-select-all">
        全选 (最多{{ max }}只)
      </button>
      <button @click="clearAll" class="btn-clear">
        清除选择
      </button>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { storeToRefs } from 'pinia'
import { useMyCatStore } from '../../stores/myCat'

interface Props {
  modelValue: string[]
  max?: number
}

const props = withDefaults(defineProps<Props>(), {
  max: 5
})

const emit = defineEmits<{
  'update:modelValue': [value: string[]]
}>()

const catStore = useMyCatStore()
const { cats } = storeToRefs(catStore)
const selected = ref<string[]>([...props.modelValue])
const loading = ref(true)

onMounted(async () => {
  try {
    await catStore.fetchCats()
  } catch (err) {
    console.error('[CatMultiSelector] Error fetching cats:', err)
  } finally {
    loading.value = false
  }
})

function toggleCat(catId: string) {
  const index = selected.value.indexOf(catId)
  if (index > -1) {
    selected.value = selected.value.filter(id => id !== catId)
  } else {
    if (selected.value.length < props.max) {
      selected.value = [...selected.value, catId]
    }
  }
  emit('update:modelValue', selected.value)
}
</script>
```

- **数量限制**：通过 max prop 控制最大选择数量
- **状态管理**：从 myCat store 获取猫咪列表
- **全选/清除**：快捷操作按钮

```typescript
export async function getBatchAnalysis(req: Request, res: Response, next: NextFunction) {
  try {
    const userId = (req as any).user?.userId
    const { catIds } = req.body

    if (!Array.isArray(catIds) || catIds.length < 2 || catIds.length > 5) {
      return res.status(400).json(errorResponse('请选择2-5只猫咪进行对比'))
    }

    const results: Record<string, any> = {}

    for (const catId of catIds) {
      const analysis = await analyzeCatWeight(catId, userId)
      if (analysis) {
        results[catId] = analysis
      }
    }

    res.json(successResponse(results, '批量分析成功'))
  } catch (error: any) {
    next(error)
  }
}
```

```vue
  const option: EChartsOption = {
    // ... ECharts 配置
```

#### V3 关键修复：时序问题

**问题**：在 V2 版本中，图表不显示是因为：
1. `initChart()` 在 `loading.value = false` 之前调用
2. 此时 DOM 中图表容器还未渲染（v-if 条件不满足）
3. `chartRef.value` 为 null，ECharts 无法初始化

**解决方案**：
```typescript
// V2 错误顺序
try {
  weightHistory.value = response.data
  await nextTick()
  initChart()  // 此时 loading 还是 true！
} finally {
  loading.value = false  // 太晚了
}

// V3 正确顺序
try {
  weightHistory.value = response.data
  loading.value = false  // 先关闭 loading
  await nextTick()       // 等待 DOM 更新
  initChart()            // 现在 chartRef 存在了
} catch (err) {
  loading.value = false
}
```

#### 容器尺寸检查

为防止容器尺寸为 0 的情况，添加了检查和重试机制：

```typescript
function initChart() {
  if (!chartRef.value) return

  // 检查容器尺寸
  const rect = chartRef.value.getBoundingClientRect()
  if (rect.width === 0 || rect.height === 0) {
    // 延迟重试
    setTimeout(() => {
      if (chartRef.value) initChart()
    }, 100)
    return
  }

  chart = echarts.init(chartRef.value)
  updateChart()
}
```

```typescript
export const useAuthStore = defineStore('auth', () => {
  const token = ref<string | null>(null)
  const user = ref<User | null>(null)

  // 初始化认证状态
  function initAuth() {
    const savedToken = localStorage.getItem('auth_token')
    const savedUser = localStorage.getItem('auth_user')

    if (savedToken && savedUser) {
      token.value = savedToken
      user.value = JSON.parse(savedUser)

      // 异步验证 token 有效性
      fetchCurrentUser().catch(() => {
        // Token 无效，清除认证信息
        clearAuth()
      })
    }
  }

  // 获取当前用户
  async function fetchCurrentUser(): Promise<boolean> {
    if (!token.value) return false

    try {
      const response = await getCurrentUser()
      if (response.success && response.data) {
        user.value = response.data
        localStorage.setItem('auth_user', JSON.stringify(response.data))
        return true
      }
      clearAuth()
      return false
    } catch (err) {
      clearAuth()
      return false
    }
  }

  return { token, user, initAuth, fetchCurrentUser }
})
```

```typescript
// 获取猫咪体重历史记录（用于趋势图）
export async function getCatWeightHistory(catId: string, userId: string) {
  // 1. 验证猫咪存在且属于用户
  const cat = await prisma.cat.findFirst({
    where: { id: catId, userId, isActive: true }
  })

  if (!cat) return null

  // 2. 获取所有成长记录
  const allRecords = await prisma.petRecord.findMany({
    where: { catId },
    orderBy: { recordDate: 'asc' }
  })

  // 3. 过滤有体重的记录
  const records = allRecords
    .filter(r => r.weight !== null && r.weight !== undefined)
    .map(record => ({
      date: record.recordDate.toISOString().split('T')[0],
      weight: record.weight!,
      notes: record.notes
    }))

  // 4. 如果猫咪档案有初始体重，且没有更早记录，添加初始点
  if (cat.weight) {
    const hasEarlierRecord = records.length > 0 &&
      new Date(records[0].date) <= cat.createdAt

    if (!hasEarlierRecord) {
      records.unshift({
        date: cat.createdAt.toISOString().split('T')[0],
        weight: cat.weight,
        notes: '初始体重'
      })
    }
  }

  return records
}
```

```typescript
function generateWeightRecords(userId: string, catId: string, cat: any) {
  const records = []
  const birthDate = new Date(cat.birthDate)
  const now = new Date()
  let recordDate = new Date(birthDate)
  recordDate.setDate(recordDate.getDate() + 56) // 从8周开始

  const baseWeight = cat.weight || 1.0
  const maxWeight = cat.gender === 'male' ? 6.0 : 5.0

  // 幼猫期（2-6月）每2周记录一次
  while (recordDate.getTime() < birthDate.getTime() + 180 * 24 * 60 * 60 * 1000) {
    const age = Math.floor((recordDate.getTime() - birthDate.getTime()) /
      (24 * 60 * 60 * 1000 * 7))
    const { ageWeeks, ageMonths } = calculateAge(birthDate, recordDate)

    baseWeight += 0.08 // 每周约增加80g

    records.push({
      userId,
      catId,
      petName: cat.name,
      recordDate: new Date(recordDate),
      weight: Math.min(baseWeight, maxWeight),
      ageWeeks,
      ageMonths,
      photoUrl: '',
      notes: `体重${baseWeight.toFixed(1)}kg，健康成长中`
    })

    recordDate.setDate(recordDate.getDate() + 14)
  }

  // 成年期（12月后）每月记录一次
  while (recordDate.getTime() < now.getTime()) {
    baseWeight += (Math.random() - 0.5) * 0.1
    const { ageWeeks, ageMonths } = calculateAge(birthDate, recordDate)

    records.push({
      userId,
      catId,
      petName: cat.name,
      recordDate: new Date(recordDate),
      weight: Math.max(3, Math.min(baseWeight, maxWeight)),
      ageWeeks,
      ageMonths,
      photoUrl: '',
      notes: '定期体检，健康状况良好'
    })

    recordDate.setMonth(recordDate.getMonth() + 1)
  }

  return records
}
```

### 常见问题排查

#### 图表不显示

1. **检查数据是否返回**
   ```
   [WeightTrend] weightHistory.length: 0
   ```

   → 表示没有体重记录，需要先添加

3. **检查认证状态**

1. **401 错误**
